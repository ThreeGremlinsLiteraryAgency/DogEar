select p.proname as function_name, pg_get_function_identity_arguments(p.oid) as arguments, pg_get_function_result(p.oid) as returns from pg_proc p join pg_namespace n on n.oid = p.pronamespace where n.nspname = 'public' and p.proname in ( 'get_or_create_club_ballot', 'publish_club_ballot', 'close_club_ballot', 'cast_club_vote' ) order by p.proname;
